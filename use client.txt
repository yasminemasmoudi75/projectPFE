  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from "@/components/ui/select"
import {
  Dialog,
  DialogContent,
  DialogDescription,
  DialogFooter,
  DialogHeader,
  DialogTitle,
} from "@/components/ui/dialog"
import { Label } from "@/components/ui/label"
import { Textarea } from "@/components/ui/textarea"

const MOCK_TIERS = [
  { IDTiers: "1", Raisoc: "TechCorp SA" },
  { IDTiers: "2", Raisoc: "InnovatePlus" },
  { IDTiers: "3", Raisoc: "GlobalTech" },
  { IDTiers: "4", Raisoc: "StartupXYZ" },
  { IDTiers: "5", Raisoc: "MegaCorp" },
]

const MOCK_PROJETS = [
  { ID_Projet: "1", Nom_Projet: "Projet Alpha", IDTiers: "1" },
  { ID_Projet: "2", Nom_Projet: "Projet Beta", IDTiers: "2" },
  { ID_Projet: "3", Nom_Projet: "Projet Gamma", IDTiers: "3" },
]

const FILTER_TYPES = [
  { name: "Tous", icon: SlidersHorizontal, type: "All" },
  { name: "Appels", icon: Phone, type: "Appel" },
  { name: "Emails", icon: Mail, type: "Email" },
  { name: "Reunions", icon: Users, type: "Reunion" },
  { name: "Visites", icon: MapPin, type: "Visite" },
  { name: "Notes", icon: FileText, type: "Note" },
]

export default function ActivitesPage() {
  const [filterType, setFilterType] = useState("All")
  const [searchTerm, setSearchTerm] = useState("")
  const [isAddModalOpen, setIsAddModalOpen] = useState(false)
  const [selectedTier, setSelectedTier] = useState("")
  const [selectedProjet, setSelectedProjet] = useState("")
  const [dateFrom, setDateFrom] = useState("")
  const [dateTo, setDateTo] = useState("")
  const [refreshing, setRefreshing] = useState(false)
  const [newActivity, setNewActivity] = useState({
    Type_Activite: "Appel",
    Description: "",
    Date_Activite: new Date().toISOString().slice(0, 16),
    Statut: "Planifie",
    IDTiers: "",
    ID_Projet: "",
  })

  const getActivityIcon = (type: string) => {
    switch (type?.toLowerCase()) {
      case "appel":
        return <Phone className="h-5 w-5" />
      case "reunion":
        return <Users className="h-5 w-5" />
      case "visite":
        return <MapPin className="h-5 w-5" />
      case "email":
        return <Mail className="h-5 w-5" />
      case "note":
        return <FileText className="h-5 w-5" />
      default:
        return <MessageCircle className="h-5 w-5" />
    }
  }

  const getTypeStyles = (type: string) => {
    switch (type?.toLowerCase()) {
      case "appel":
        return { bg: "bg-sky-50", text: "text-sky-500", border: "border-sky-100" }
      case "reunion":
        return { bg: "bg-violet-50", text: "text-violet-500", border: "border-violet-100" }
      case "visite":
        return { bg: "bg-emerald-50", text: "text-emerald-500", border: "border-emerald-100" }
      case "email":
        return { bg: "bg-cyan-50", text: "text-cyan-500", border: "border-cyan-100" }
      case "note":
        return { bg: "bg-amber-50", text: "text-amber-500", border: "border-amber-100" }
      default:
        return { bg: "bg-slate-50", text: "text-slate-500", border: "border-slate-100" }
    }
  }

  const getStatusStyles = (statut: string) => {
    switch (statut) {
      case "Termine":
        return { bg: "bg-emerald-50", text: "text-emerald-500" }
      case "Planifie":
        return { bg: "bg-amber-50", text: "text-amber-500" }
      case "En cours":
        return { bg: "bg-sky-50", text: "text-sky-500" }
      default:
        return { bg: "bg-slate-50", text: "text-slate-500" }
    }
  }

  const filteredActivites = useMemo(() => {
    return MOCK_ACTIVITES.filter((a) => {
      const matchesType =
        filterType === "All" ||
        a.Type_Activite?.toLowerCase() === filterType.toLowerCase()
      const matchesSearch =
        (a.Description?.toLowerCase() || "").includes(searchTerm.toLowerCase()) ||
        (a.Type_Activite?.toLowerCase() || "").includes(searchTerm.toLowerCase()) ||
        (a.tier?.Raisoc?.toLowerCase() || "").includes(searchTerm.toLowerCase())
      return matchesType && matchesSearch
    })
  }, [filterType, searchTerm])

  const activityStats = useMemo(() => {
    const total = filteredActivites.length
    let done = 0
    let planned = 0
    let inProgress = 0

    filteredActivites.forEach((a) => {
      if (a.Statut === "Termine") done += 1
      if (a.Statut === "Planifie") planned += 1
      if (a.Statut === "En cours") inProgress += 1
    })

    return { total, done, planned, inProgress }
  }, [filteredActivites])

  const handleRefresh = () => {
    setRefreshing(true)
    setTimeout(() => setRefreshing(false), 1000)
  }

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault()
    setIsAddModalOpen(false)
    setNewActivity({
      Type_Activite: "Appel",
      Description: "",
      Date_Activite: new Date().toISOString().slice(0, 16),
      Statut: "Planifie",
      IDTiers: "",
      ID_Projet: "",
    })
  }

  return (
    <div className="min-h-screen bg-background">
      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-8">
        {/* Header */}
        <Card className="mb-8 border-none shadow-sm bg-gradient-to-r from-sky-50 via-white to-violet-50">
          <CardContent className="p-8">
            <div className="flex flex-col lg:flex-row lg:items-center justify-between gap-6">
              <div>
                <div className="inline-flex items-center gap-1.5 px-3 py-1 rounded-full bg-sky-100 text-sky-500 text-xs font-medium mb-3">
                  <Sparkles className="h-3 w-3" />
                  Journal CRM
                </div>
                <h1 className="text-3xl font-bold text-foreground tracking-tight">
                  Journal d&apos;Activites
                </h1>
                <p className="text-muted-foreground mt-1 flex items-center gap-2">
                  <Clock className="h-4 w-4 text-sky-500" />
                  Historique des interactions clients
                </p>
              </div>

              <div className="flex items-center gap-3">
                <Button
                  variant="outline"
                  size="icon"
                  onClick={handleRefresh}
                  className={`rounded-xl border-border hover:bg-secondary ${
                    refreshing ? "animate-spin" : ""
                  }`}
                >
                  <RefreshCw className="h-4 w-4" />
                </Button>
                <Button
                  onClick={() => setIsAddModalOpen(true)}
                  className="bg-sky-500 hover:bg-sky-400 text-white rounded-xl shadow-md shadow-sky-200/50"
                >
                  <Plus className="h-4 w-4 mr-2" />
                  Nouvelle Activite
                </Button>
              </div>
            </div>
          </CardContent>
        </Card>

        {/* Filters */}
        <Card className="mb-6 shadow-sm">
          <CardHeader className="border-b border-border bg-secondary/30 py-4">
            <div className="flex items-center justify-between">
              <CardTitle className="text-xs font-semibold text-muted-foreground uppercase tracking-widest">
                Filtres & Segmentation
              </CardTitle>
              <span className="text-xs text-muted-foreground flex items-center gap-1">
                <Filter className="h-3 w-3" />
                Vue filtree en temps reel
              </span>
            </div>
          </CardHeader>
          <CardContent className="p-6">
            {/* Channel Filters */}
            <div className="mb-6">
              <h4 className="text-xs font-medium text-muted-foreground mb-3 uppercase tracking-wider">
                Canal
              </h4>
              <div className="flex flex-wrap gap-2">
                {FILTER_TYPES.map((item) => (
                  <button
                    key={item.type}
                    onClick={() => setFilterType(item.type)}
                    className={`flex items-center gap-2 px-4 py-2 rounded-xl text-xs font-medium transition-all border ${
                      filterType === item.type
                        ? "bg-sky-50 border-sky-200 text-sky-500 shadow-sm"
                        : "bg-card border-border text-muted-foreground hover:bg-secondary"
                    }`}
                  >
                    <item.icon className="h-4 w-4" />
                    <span>{item.name}</span>
                  </button>
                ))}
              </div>
            </div>

            {/* Additional Filters */}
            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
              <div>
                <Label className="text-xs text-muted-foreground uppercase tracking-wider mb-2 block">
                  Client
                </Label>
                <Select value={selectedTier} onValueChange={setSelectedTier}>
                  <SelectTrigger className="bg-card border-border rounded-xl">
                    <SelectValue placeholder="Tous les clients" />
                  </SelectTrigger>
                  <SelectContent>
                    <SelectItem value="all">Tous les clients</SelectItem>
                    {MOCK_TIERS.map((t) => (
                      <SelectItem key={t.IDTiers} value={t.IDTiers}>
                        {t.Raisoc}
                      </SelectItem>
                    ))}
                  </SelectContent>
                </Select>
              </div>

              <div>
                <Label className="text-xs text-muted-foreground uppercase tracking-wider mb-2 block">
                  Projet
                </Label>
                <Select value={selectedProjet} onValueChange={setSelectedProjet}>
                  <SelectTrigger className="bg-card border-border rounded-xl">
                    <SelectValue placeholder="Tous les projets" />
                  </SelectTrigger>
                  <SelectContent>
                    <SelectItem value="all">Tous les projets</SelectItem>
                    {MOCK_PROJETS.map((p) => (
                      <SelectItem key={p.ID_Projet} value={p.ID_Projet}>
                        {p.Nom_Projet}
                      </SelectItem>
                    ))}
                  </SelectContent>
                </Select>
              </div>

              <div className="sm:col-span-2">
                <Label className="text-xs text-muted-foreground uppercase tracking-wider mb-2 block">
                  Periode
                </Label>
                <div className="flex gap-2">
                  <Input
                    type="date"
                    value={dateFrom}
                    onChange={(e) => setDateFrom(e.target.value)}
                    className="bg-card border-border rounded-xl"
                    placeholder="Du"
                  />
                  <Input
                    type="date"
                    value={dateTo}
                    onChange={(e) => setDateTo(e.target.value)}
                    className="bg-card border-border rounded-xl"
                    placeholder="Au"
                  />
                </div>
              </div>
            </div>
          </CardContent>
        </Card>

        {/* Stats Cards */}
        <div className="grid grid-cols-2 lg:grid-cols-4 gap-4 mb-6">
          {[
            {
              label: "Total Activites",
              value: activityStats.total,
              icon: Activity,
              color: "sky",
            },
            {
              label: "Terminees",
              value: activityStats.done,
              icon: CheckCircle2,
              color: "emerald",
            },
            {
              label: "Planifiees",
              value: activityStats.planned,
              icon: Calendar,
              color: "amber",
            },
            {
              label: "En cours",
              value: activityStats.inProgress,
              icon: TrendingUp,
              color: "violet",
            },
          ].map((stat, i) => {
            const colorMap = {
              sky: { bg: "bg-sky-50", text: "text-sky-500", bar: "bg-sky-400" },
              emerald: { bg: "bg-emerald-50", text: "text-emerald-500", bar: "bg-emerald-400" },
              amber: { bg: "bg-amber-50", text: "text-amber-500", bar: "bg-amber-400" },
              violet: { bg: "bg-violet-50", text: "text-violet-500", bar: "bg-violet-400" },
            }
            const colors = colorMap[stat.color as keyof typeof colorMap]

            return (
              <Card
                key={i}
                className="shadow-sm border-border hover:shadow-md transition-shadow group"
              >
                <CardContent className="p-5">
                  <div className="flex items-start justify-between">
                    <div className="flex-1">
                      <p className="text-[10px] font-semibold text-muted-foreground uppercase tracking-widest mb-1">
                        {stat.label}
                      </p>
                      <h3 className="text-2xl font-bold text-foreground">
                        {stat.value}
                      </h3>
                    </div>
                    <div
                      className={`${colors.bg} p-2.5 rounded-xl group-hover:scale-110 transition-transform`}
                    >
                      <stat.icon className={`h-5 w-5 ${colors.text}`} />
                    </div>
                  </div>
                  <div className={`h-1 ${colors.bar} mt-4 rounded-full`}></div>
                </CardContent>
              </Card>
            )
          })}
        </div>

        {/* Search Bar */}
        <Card className="mb-6 shadow-sm">
          <CardContent className="p-4">
            <div className="flex flex-col sm:flex-row items-center justify-between gap-4">
              <div className="flex items-center gap-3">
                <div
                  className={`h-10 w-10 rounded-xl flex items-center justify-center text-sm font-bold ${
                    filterType === "All"
                      ? "bg-slate-100 text-slate-500"
                      : `${getTypeStyles(filterType).bg} ${getTypeStyles(filterType).text}`
                  }`}
                >
                  {filterType === "All" ? (
                    <SlidersHorizontal className="h-5 w-5" />
                  ) : (
                    getActivityIcon(filterType)
                  )}
                </div>
                <div>
                  <h2 className="text-sm font-semibold text-foreground">
                    Journal {filterType === "All" ? "Omnicanal" : filterType}
                  </h2>
                  <p className="text-xs text-muted-foreground">
                    {filteredActivites.length} interaction(s)
                  </p>
                </div>
              </div>
              <div className="relative w-full sm:w-80">
                <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
                <Input
                  type="text"
                  placeholder="Rechercher dans les notes, les types..."
                  value={searchTerm}
                  onChange={(e) => setSearchTerm(e.target.value)}
                  className="pl-10 rounded-xl border-border bg-card"
                />
              </div>
            </div>
          </CardContent>
        </Card>

        {/* Timeline */}
        <Card className="shadow-sm">
          <CardHeader className="border-b border-border bg-secondary/30 py-4">
            <div className="flex items-center justify-between">
              <CardTitle className="text-sm font-semibold text-foreground">
                Flux d&apos;activites
              </CardTitle>
              <span className="text-xs text-muted-foreground flex items-center gap-1">
                <Clock className="h-4 w-4" />
                Triees par date d&apos;interaction
              </span>
            </div>
          </CardHeader>
          <CardContent className="p-6">
            <div className="space-y-4">
              {filteredActivites.length === 0 ? (
                <div className="border-2 border-dashed border-border rounded-2xl px-8 py-16 text-center bg-secondary/30">
                  <RefreshCw className="h-10 w-10 mx-auto text-muted-foreground mb-4" />
                  <h3 className="text-sm font-semibold text-foreground">
                    Aucune activite
                  </h3>
                  <p className="text-xs text-muted-foreground mt-1">
                    Ajustez vos filtres ou enregistrez une nouvelle interaction
                    commerciale.
                  </p>
                </div>
              ) : (
                filteredActivites.map((activite) => {
                  const typeStyles = getTypeStyles(activite.Type_Activite)
                  const statusStyles = getStatusStyles(activite.Statut)

                  return (
                    <div
                      key={activite.ID_Activite}
                      className="group border border-border rounded-xl bg-card px-6 py-5 hover:border-sky-200 hover:shadow-md transition-all"
                    >
                      <div className="flex flex-col sm:flex-row sm:items-start gap-4">
                        <div
                          className={`flex-shrink-0 h-12 w-12 rounded-xl flex items-center justify-center ${typeStyles.bg} ${typeStyles.text} group-hover:scale-110 transition-transform`}
                        >
                          {getActivityIcon(activite.Type_Activite)}
                        </div>
                        <div className="flex-1 min-w-0">
                          <div className="flex flex-wrap items-center gap-2 mb-2">
                            <h3 className="text-base font-semibold text-foreground">
                              {activite.Type_Activite}
                            </h3>
                            <span
                              className={`inline-flex items-center px-2.5 py-1 rounded-lg text-[10px] font-medium uppercase tracking-wider ${statusStyles.bg} ${statusStyles.text}`}
                            >
                              {activite.Statut}
                            </span>
                          </div>
                          <p className="text-sm text-muted-foreground mb-3 leading-relaxed">
                            {activite.Description || "Aucune description"}
                          </p>
                          <div className="flex flex-wrap items-center gap-x-4 gap-y-2 text-xs text-muted-foreground">
                            <span className="flex items-center gap-1.5">
                              <Clock className="h-4 w-4" />
                              {new Date(activite.Date_Activite).toLocaleTimeString(
                                "fr-FR",
                                { hour: "2-digit", minute: "2-digit" }
                              )}
                            </span>
                            <span className="flex items-center gap-1.5">
                              <Calendar className="h-4 w-4" />
                              {new Date(activite.Date_Activite).toLocaleDateString(
                                "fr-FR"
                              )}
                            </span>
                            <span className="flex items-center gap-1.5">
                              <User className="h-4 w-4" />
                              {activite.utilisateur?.FullName || "Collaborateur"}
                            </span>
                            {activite.tier && (
                              <span className="flex items-center gap-1.5">
                                <Building2 className="h-4 w-4" />
                                {activite.tier.Raisoc || "Client"}
                              </span>
                            )}
                          </div>
                        </div>
                        <div className="flex items-center gap-2 sm:self-center">
                          <Button
                            variant="outline"
                            size="sm"
                            className="text-xs font-medium text-sky-500 hover:text-white hover:bg-sky-500 rounded-lg border-sky-200"
                          >
                            Voir details
                          </Button>
                        </div>
                      </div>
                    </div>
                  )
                })
              )}
            </div>
          </CardContent>
        </Card>

        {/* Modal - New Activity */}
        <Dialog open={isAddModalOpen} onOpenChange={setIsAddModalOpen}>
          <DialogContent className="sm:max-w-lg">
            <DialogHeader>
              <DialogTitle>Nouvelle activite</DialogTitle>
              <DialogDescription>
                Enregistrer une interaction client
              </DialogDescription>
            </DialogHeader>

            <form onSubmit={handleSubmit} className="space-y-5">
              <div className="grid grid-cols-2 gap-4">
                <div>
                  <Label className="text-xs text-muted-foreground uppercase tracking-wider mb-2 block">
                    Canal
                  </Label>
                  <Select
                    value={newActivity.Type_Activite}
                    onValueChange={(value) =>
                      setNewActivity({ ...newActivity, Type_Activite: value })
                    }
                  >
                    <SelectTrigger className="rounded-xl">
                      <SelectValue />
                    </SelectTrigger>
                    <SelectContent>
                      <SelectItem value="Appel">Appel</SelectItem>
                      <SelectItem value="Email">Email</SelectItem>
                      <SelectItem value="Visite">Visite</SelectItem>
                      <SelectItem value="Reunion">Reunion</SelectItem>
                      <SelectItem value="Note">Note</SelectItem>
                    </SelectContent>
                  </Select>
                </div>
                <div>
                  <Label className="text-xs text-muted-foreground uppercase tracking-wider mb-2 block">
                    Statut
                  </Label>
                  <Select
                    value={newActivity.Statut}
                    onValueChange={(value) =>
                      setNewActivity({ ...newActivity, Statut: value })
                    }
                  >
                    <SelectTrigger className="rounded-xl">
                      <SelectValue />
                    </SelectTrigger>
                    <SelectContent>
                      <SelectItem value="Planifie">Planifie</SelectItem>
                      <SelectItem value="En cours">En cours</SelectItem>
                      <SelectItem value="Termine">Termine</SelectItem>
                      <SelectItem value="Annule">Annule</SelectItem>
                    </SelectContent>
                  </Select>
                </div>
              </div>

              <div>
                <Label className="text-xs text-muted-foreground uppercase tracking-wider mb-2 block">
                  Description
                </Label>
                <Textarea
                  value={newActivity.Description}
                  onChange={(e) =>
                    setNewActivity({ ...newActivity, Description: e.target.value })
                  }
                  rows={3}
                  placeholder="Details de l'interaction..."
                  className="rounded-xl resize-none"
                />
              </div>

              <div>
                <Label className="text-xs text-muted-foreground uppercase tracking-wider mb-2 block">
                  Date et heure
                </Label>
                <Input
                  type="datetime-local"
                  value={newActivity.Date_Activite}
                  onChange={(e) =>
                    setNewActivity({ ...newActivity, Date_Activite: e.target.value })
                  }
                  className="rounded-xl"
                />
              </div>

              <div className="grid grid-cols-2 gap-4">
                <div>
                  <Label className="text-xs text-muted-foreground uppercase tracking-wider mb-2 block">
                    Client
                  </Label>
                  <Select
                    value={newActivity.IDTiers}
                    onValueChange={(value) =>
                      setNewActivity({ ...newActivity, IDTiers: value, ID_Projet: "" })
                    }
                  >
                    <SelectTrigger className="rounded-xl">
                      <SelectValue placeholder="Selectionner" />
                    </SelectTrigger>
                    <SelectContent>
                      {MOCK_TIERS.map((t) => (
                        <SelectItem key={t.IDTiers} value={t.IDTiers}>
                          {t.Raisoc}
                        </SelectItem>
                      ))}
                    </SelectContent>
                  </Select>
                </div>
                <div>
                  <Label className="text-xs text-muted-foreground uppercase tracking-wider mb-2 block">
                    Projet lie
                  </Label>
                  <Select
                    value={newActivity.ID_Projet}
                    onValueChange={(value) =>
                      setNewActivity({ ...newActivity, ID_Projet: value })
                    }
                  >
                    <SelectTrigger className="rounded-xl">
                      <SelectValue placeholder="Selectionner" />
                    </SelectTrigger>
                    <SelectContent>
                      {MOCK_PROJETS.map((p) => (
                        <SelectItem key={p.ID_Projet} value={p.ID_Projet}>
                          {p.Nom_Projet}
                        </SelectItem>
                      ))}
                    </SelectContent>
                  </Select>
                </div>
              </div>

              <DialogFooter className="gap-3 pt-2">
                <Button
                  type="button"
                  variant="outline"
                  onClick={() => setIsAddModalOpen(false)}
                  className="flex-1 rounded-xl"
                >
                  Annuler
                </Button>
                <Button
                  type="submit"
                  className="flex-1 bg-sky-500 hover:bg-sky-400 text-white rounded-xl shadow-md"
                >
                  Enregistrer
                </Button>
              </DialogFooter>
            </form>
          </DialogContent>
        </Dialog>
      </div>
    </div>
  )
}
